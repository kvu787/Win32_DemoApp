cmake_minimum_required(VERSION 3.24)
project(Win32_DemoApp LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows 11 x64 is the supported platform.")
endif()
add_executable(Win32_DemoApp WIN32 Source/Main.cpp Source/Gallery.cpp Source/TextSearch.cpp Source/Regression.cpp Source/App.cpp Source/App.rc)
target_compile_features(Win32_DemoApp PRIVATE cxx_std_20)
target_compile_definitions(Win32_DemoApp PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(Win32_DemoApp PRIVATE /W4 /permissive- /utf-8 /EHsc)
set_property(TARGET Win32_DemoApp PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(Win32_DemoApp PRIVATE comctl32 comdlg32 shell32 ole32 uuid uxtheme dwmapi)
enable_testing()
add_test(NAME NativeControlAudit COMMAND Win32_DemoApp --self-test --report ${CMAKE_BINARY_DIR}/Audit.txt)
